101inline bool release_slot(parking_lot &lot, int slot_id)
102{
103 if (slot_id < 1 || slot_id > PARKING_CAPACITY)
104 {
105 println(format("Invalid Slot {}. Must be between 1 and {}.", slot_id, PARKING_CAPACITY));
106 return false;
107 }
108
109 parking_slot &slot = lot.slots[slot_id - 1];
110 if (!slot.is_occupied)
111 {
112 println(format("Slot {} is already empty.", slot_id));
113 return false;
114 }
115
116 // Save departure to history queue
117 departure_record record = {slot.vehicle_id, slot.slot_id};
118 lot.departure_history.enqueue(record);
119
120 println(format("Slot {} released. Vehicle '{}' departed.", slot_id, slot.vehicle_id));
121
122 // Reset slot
123 slot.is_occupied = false;
124 slot.vehicle_id = "";
125 return true;
126}
